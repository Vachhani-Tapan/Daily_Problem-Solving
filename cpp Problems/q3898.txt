class Solution {
public:
    vector<int> findDegrees(vector<vector<int>>& matrix) {
        vector<int> v1;
        for(int i = 0 ; i < matrix.size() ; i++){
            int count = 0;
            for(int j = 0 ; j < matrix[i].size() ; j++){
                if(matrix[i][j] == 1){
                    count++;
                }
            }
            v1.push_back(count);
        }
        return v1;
    }
};
