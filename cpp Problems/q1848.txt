class Solution {
public:
    int getMinDistance(vector<int>& nums, int target, int start) {
        // [2,1,1,1,1,1,2]  target = 2; start = 3;
        int ans = INT_MAX;
        for (int i = start; i < nums.size(); i++) {
            if (nums[i] == target) {
                ans = min(ans, abs(i - start));
            }
        }
        for (int i = 0; i < nums.size(); i++) {
            if (nums[i] == target) {
                ans = min(ans, abs(i - start));
            }
        }
        return ans;
    }
};
